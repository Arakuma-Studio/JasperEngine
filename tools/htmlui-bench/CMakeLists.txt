# SVG library benchmark (LunaSVG / ThorVG / nanosvg). Standalone: not part of the engine or
# editor build. See README and docs/html-ui/svg-benchmark.md.
#
#   cmake -S tools/htmlui-bench -B build/htmlui-bench/native -DCMAKE_BUILD_TYPE=Release
#   cmake -S tools/htmlui-bench -B build/htmlui-bench/size -DBENCH_SIZE=ON
#   emcmake cmake -S tools/htmlui-bench -B build/htmlui-bench/wasm -DCMAKE_BUILD_TYPE=Release
#
# Every library is compiled from source here with the same compiler and flags (-O2, or -Os
# for the size probes), so the comparison does not depend on each project's own build system.
cmake_minimum_required(VERSION 3.16)
project(htmlui_bench C CXX)

get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LUNASVG_DIR "/tmp/htmlui-src/lunasvg" CACHE PATH "LunaSVG source tree")
set(THORVG_DIR "/tmp/htmlui-src/thorvg" CACHE PATH "ThorVG source tree")
set(NANOSVG_DIR "${REPO_ROOT}/engine/libs/nanosvg" CACHE PATH "nanosvg headers")
set(STB_DIR "${REPO_ROOT}/engine/libs/stb" CACHE PATH "stb headers (stb_image_write)")
option(BENCH_SIZE "Build the binary-size probes (-Os, gc-sections, stripped) instead of svgbench" OFF)
option(THORVG_AVX "ThorVG AVX2 raster paths (x86-64 native only)" ON)
option(THORVG_THREADS "ThorVG task scheduler support (svgbench still runs single-threaded unless --threads)" ON)
set(THORVG_BENCH_NAME "thorvg" CACHE STRING "Label of the ThorVG backend in the results")
set(THORVG_VERSION "1.2.0")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

if(BENCH_SIZE)
    set(BENCH_OPT "-Os")
else()
    set(BENCH_OPT "-O2")
endif()
# Same optimisation for all libraries (CMake's Release default would be -O3).
set(CMAKE_C_FLAGS_RELEASE "${BENCH_OPT} -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "${BENCH_OPT} -DNDEBUG")
if(BENCH_SIZE)
    add_compile_options(-ffunction-sections -fdata-sections)
    if(EMSCRIPTEN)
        add_link_options(-Os)
    else()
        add_link_options(-Wl,--gc-sections -s)
    endif()
endif()

set(IS_X86 FALSE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|i.86" AND NOT EMSCRIPTEN)
    set(IS_X86 TRUE)
endif()

# ---------------------------------------------------------------- LunaSVG + plutovg
file(GLOB LUNASVG_SOURCES ${LUNASVG_DIR}/source/*.cpp)
file(GLOB PLUTOVG_SOURCES ${LUNASVG_DIR}/plutovg/source/*.c)
add_library(bench_lunasvg STATIC ${LUNASVG_SOURCES} ${PLUTOVG_SOURCES})
target_include_directories(bench_lunasvg PUBLIC ${LUNASVG_DIR}/include
    PRIVATE ${LUNASVG_DIR}/source ${LUNASVG_DIR}/plutovg/include ${LUNASVG_DIR}/plutovg/source)
target_compile_definitions(bench_lunasvg PUBLIC LUNASVG_BUILD_STATIC PLUTOVG_BUILD_STATIC
    PRIVATE LUNASVG_BUILD PLUTOVG_BUILD)
target_compile_options(bench_lunasvg PRIVATE -w)
if(NOT EMSCRIPTEN)
    target_link_libraries(bench_lunasvg PRIVATE m)
endif()

# ---------------------------------------------------------------- ThorVG (CPU renderer)
# add_thorvg(<target> <loaders...>): loaders among svg png jpg ttf lottie
function(add_thorvg target)
    set(loaders ${ARGN})
    set(src ${THORVG_DIR}/src)
    file(GLOB sources ${src}/common/*.cpp ${src}/renderer/*.cpp ${src}/renderer/cpu_engine/*.cpp ${src}/loaders/raw/*.cpp)
    set(includes ${THORVG_DIR}/inc ${src}/common ${src}/renderer ${src}/renderer/cpu_engine ${src}/loaders/raw)
    set(THORVG_SVG_LOADER_SUPPORT OFF)
    set(THORVG_PNG_LOADER_SUPPORT OFF)
    set(THORVG_JPG_LOADER_SUPPORT OFF)
    set(THORVG_SFNT_LOADER_SUPPORT OFF)
    set(THORVG_TTF_LOADER_SUPPORT OFF)
    set(THORVG_LOTTIE_LOADER_SUPPORT OFF)
    foreach(l ${loaders})
        if(l STREQUAL "svg")
            set(THORVG_SVG_LOADER_SUPPORT ON)
            set(dir svg)
        elseif(l STREQUAL "png")
            set(THORVG_PNG_LOADER_SUPPORT ON)
            set(dir png)
        elseif(l STREQUAL "jpg")
            set(THORVG_JPG_LOADER_SUPPORT ON)
            set(dir jpg)
        elseif(l STREQUAL "ttf")
            set(THORVG_SFNT_LOADER_SUPPORT ON)
            set(THORVG_TTF_LOADER_SUPPORT ON)
            set(dir sfnt)
        elseif(l STREQUAL "lottie")
            set(THORVG_LOTTIE_LOADER_SUPPORT ON)
            set(dir lottie)
        endif()
        file(GLOB lsrc ${src}/loaders/${dir}/*.cpp)
        list(APPEND sources ${lsrc})
        list(APPEND includes ${src}/loaders/${dir})
    endforeach()
    set(THORVG_THREAD_SUPPORT ${THORVG_THREADS})
    set(THORVG_FILE_IO_SUPPORT ON)
    set(THORVG_PARTIAL_RENDER_SUPPORT ON)
    set(THORVG_AVX_SUPPORT OFF)
    if(THORVG_AVX AND IS_X86)
        set(THORVG_AVX_SUPPORT ON)
    endif()
    configure_file(${CMAKE_CURRENT_SOURCE_DIR}/thorvg_config.h.in ${CMAKE_CURRENT_BINARY_DIR}/${target}/config.h)
    add_library(${target} STATIC ${sources})
    target_include_directories(${target} PUBLIC ${THORVG_DIR}/inc PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/${target} ${includes})
    target_compile_definitions(${target} PUBLIC TVG_STATIC)
    target_compile_options(${target} PRIVATE -w)
    if(THORVG_AVX_SUPPORT)
        target_compile_options(${target} PRIVATE -mavx2)
    endif()
    if(THORVG_THREADS AND NOT EMSCRIPTEN)
        find_package(Threads REQUIRED)
        target_link_libraries(${target} PUBLIC Threads::Threads)
    endif()
endfunction()

if(NOT BENCH_SIZE)
    # Same loader set as LunaSVG's built-in capabilities: SVG, embedded PNG/JPEG, TTF text.
    add_thorvg(bench_thorvg svg png jpg ttf)

    add_executable(svgbench
        src/main.cpp src/BackendLuna.cpp src/BackendThorVG.cpp src/BackendNanoSVG.cpp
        ${STB_DIR}/stb_image_write.c)
    target_include_directories(svgbench PRIVATE src ${NANOSVG_DIR} ${STB_DIR})
    target_compile_definitions(svgbench PRIVATE THORVG_BENCH_NAME="${THORVG_BENCH_NAME}")
    target_compile_options(svgbench PRIVATE -Wall -Wextra)
    set_source_files_properties(src/BackendNanoSVG.cpp ${STB_DIR}/stb_image_write.c PROPERTIES COMPILE_OPTIONS "-w")
    target_link_libraries(svgbench PRIVATE bench_lunasvg bench_thorvg)
    if(EMSCRIPTEN)
        target_link_options(svgbench PRIVATE -sNODERAWFS=1 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1
            -sSTACK_SIZE=4MB -sEXIT_RUNTIME=1)
    endif()
else()
    # Binary-size probes: each program loads the SVG given on the command line, renders it at
    # 64 px and writes the pixels, so the linker keeps everything a real user would need.
    add_thorvg(size_thorvg_svg svg)
    add_thorvg(size_thorvg_svgfull svg png jpg ttf)
    add_thorvg(size_thorvg_svglottie svg png jpg ttf lottie)

    add_executable(size_baseline src/size/baseline.cpp)
    add_executable(size_lunasvg src/size/lunasvg.cpp)
    target_link_libraries(size_lunasvg PRIVATE bench_lunasvg)
    add_executable(size_nanosvg src/size/nanosvg.cpp)
    target_include_directories(size_nanosvg PRIVATE ${NANOSVG_DIR})
    target_compile_options(size_nanosvg PRIVATE -w)
    foreach(v svg svgfull svglottie)
        add_executable(size_thorvg_${v}_probe src/size/thorvg.cpp)
        target_link_libraries(size_thorvg_${v}_probe PRIVATE size_thorvg_${v})
    endforeach()
    if(EMSCRIPTEN)
        foreach(t size_baseline size_lunasvg size_nanosvg size_thorvg_svg_probe size_thorvg_svgfull_probe size_thorvg_svglottie_probe)
            target_link_options(${t} PRIVATE -sNODERAWFS=1 -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1)
        endforeach()
    endif()
endif()
